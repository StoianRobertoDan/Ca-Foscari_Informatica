#include<iostream>
#include<string>

using namespace std;

class Dict{
  public:
    Dict();
    ~Dict();

    int& at(string pos);
    void print() const;
    bool operator<=(const Dict& d) const;
    bool operator==(const Dict& d) const;
    Dict& operator+=(const Dict& d);

  private:
    struct Cella{
      string key;
      int val;
      Cella* next;
    };

    typedef Cella* pCella;

    pCella head;



};

//Default constructor. Do not change
Dict::Dict() {
  head = nullptr;
}

//Destructor. Do not change
Dict::~Dict() {
  pCella pc;
  while (head) {
    pc = head;
    head = head->next;
    delete pc;
  }
}

//Print elements. Do not change
void Dict:: print() const {
  cout<<"OUTPUT"<<endl;
  pCella pc = head;

  while (pc) {
    cout<<pc->key<<":"<<pc->val<<endl;
    pc = pc->next;
  }
  cout<<"---"<<endl;
}


// Exercise 1
// Implement: the method 'at' returns a reference to the 'value'
// field of the node whose key is equal to 'pos', if it exists;
// otherwise, it creates a new node at the end of the list
// with 'pos' as the key and 0 as the value,
// and returns a reference to the 'value' field of this new node.
int& Dict::at(string pos) {
    if(head){
        auto pt = head;
        if(head->key == pos) return head->val;
        while(pt->next != nullptr) {
            if(pt->key == pos) return pt->val;
            pt = pt->next;
        }
        if(pt->key == pos) return pt->val;
        auto c = new Cella{pos, 0, nullptr};
        pt->next = c;
        return c->val;
    }
    else{
        head = new Cella{pos, 0, nullptr};
        return head->val;
    }
}

// Returns true if *this is a subset of the elements in d,
// that is, all key-value pairs in *this are also present
// in Dict d (even if in a different order)
bool Dict::operator<=(const Dict& d) const {
    if(head and d.head){
        auto pt = head;
        bool minore = true;
        while(pt and minore){
            auto ptd = d.head;
            bool found = false;
            while(ptd and !found){
                if(pt->key == ptd->key){
                    found = true;
                    if(pt->val != ptd->val) minore = false;
                }
                ptd = ptd->next;
            }
            if(!found) return false;
            pt = pt->next;
        }
        return minore;
    }
    if(d.head and !head) return true;
    else return false;
}

// Exercise 2
// Returns true if *this has the same elements as d,
// even if in a different order
bool Dict::operator==(const Dict& d) const {
    if(*this <= d and d <= *this) return true;
    else return false;
}

// Exercise 3
// Adds all elements from the parameter d whose key is not
// already present in *this.
// In other words, for each element in d, if its key is already
// present in *this, it is ignored; if it is not present,
// it is added to the end of the list. 
// At the end, *this is returned.

Dict& Dict::operator+=(const Dict& d) {
    if(d.head and !(*this == d)){
        auto ptd = d.head;
        while(ptd){
            auto pt = head;
            while(pt and ptd->key != pt->key) pt = pt->next;
            if(!pt) this->at(ptd->key) = ptd->val;
            ptd = ptd->next;
        }
    }
    return *this;
}
