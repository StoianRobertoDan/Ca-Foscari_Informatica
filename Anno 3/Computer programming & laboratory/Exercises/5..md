#include<iostream>

struct Cell{
  int info;
  Cell* next;
};

typedef Cell* List;

bool findsum(List l, int sum, int add) {
    if(sum == 0 and add == 0) return true;
    else {
        if(sum == 0 or add == 0 or !l) return false;
        else{
            List head = l;
            return findsum(head->next, sum - head->info, add-1) + findsum(head->next,sum ,add);
        }
    }
}

class Primes{
private:
    std::vector<int> primes;
    
public:
    Primes(int size) {
        primes.push_back(2);
        get_prime(size);
    }

    int get_prime(int n) {
        if(n < primes.size()){
            return primes.at(n);
        }
        else{
            int num = primes.back() + 1;
            while(primes.size() <= n){
                if(is_prime(num)) primes.push_back(num);
                num++;
            }
            return primes.at(n);
        }
    }
    
    bool is_prime(int n){
        for(int i : primes) if(n % i == 0) return false;
        return true;
    }
};

#include<iostream>

struct Cell{
  int info;
  Cell* next;
};

typedef Cell* List;

void append(List& l, int elem) {
  if (l==nullptr) {
    l = new Cell{elem, nullptr};
  }
  else 
    append(l->next, elem);
}


void readlist(List& l) {
  int q, e;

  std::cin>>q;

  while (q>0) {
    std::cin>>e;
    append(l, e);
    q--;
  }
}

void destroy(List& l) {
  if (l) {
    destroy(l->next);
    delete l;
    l=nullptr;
  }
}


void printlist(List l) {
  if (l) {
    std::cout<<l->info<<std::endl;
    printlist(l->next);
  }
}

List merge(List l1, List l2) {
    List nl = nullptr, h1 = l1, h2 = l2;
    while(h1 != nullptr or h2 != nullptr){
        if(h1 == nullptr) {
            append(nl, h2->info);
            h2 = h2->next;
        }
        else if(h2 == nullptr) {
            append(nl, h1->info);
            h1 = h1->next;
        }
        else{
            if(h1->info <= h2->info) {
                append(nl, h1->info);
                h1 = h1->next;
            }
            else{
                append(nl, h2->info);
                h2 = h2->next;
            }
        }
    }
    return nl;
}
