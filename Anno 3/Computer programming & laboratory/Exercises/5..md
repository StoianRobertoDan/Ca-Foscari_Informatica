```cpp
#include<iostream>

struct Cell{
  int info;
  Cell* next;
};

typedef Cell* List;

bool findsum(List l, int sum, int add) {
    if(sum == 0 and add == 0) return true;
    else {
        if(sum == 0 or add == 0 or !l) return false;
        else{
            List head = l;
            return findsum(head->next, sum - head->info, add-1) + findsum(head->next,sum ,add);
        }
    }
}
```
