Number::Number(std::string s){
    n = convert_str;
}

Number::List Number::convert_str(std::string s){
    if(s.length() == 0 or only_zeros())
        return nullptr;
    list pc = new Cell;
    pc.info = s.back()-'0';
    pc.next = convert_str(s.substring(0, s.lenght()-1));
    return pc;
}

Number::List Number::sum(List l1, List l2, char carry){
    if(!l1 and !l2){
        if(carry != 0){
            List last = new Cell{carry, nullptr}
            return last;
        }
        else return nullptr;
    //else if(!l1)
    //else if(!l2)
    //else
    char n1 = l1 ? l1.info : 0;
    char n2 = l2 ? l2.info : 0;
    char s = n1 + n2 + carry;
    List n = new Cell{s%10, sum(l1 ? l1->next : nullptr, l2 ? l2->next : nullptr, s/10)};
    return n;
}

bool Number::operator==(Number const& x) const{
    return equal(x.n, n);
}

bool Number::equal(List l1, List l2){
    if(!l1 and !l2) return true;
    else if(!l1 or !l2) return false;   //xor is also correct
    else{
        return (l1.info == l2.info) and equal(l1->next, l2->next);
    }
}
