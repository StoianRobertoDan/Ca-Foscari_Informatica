#include<iostream>

using namespace std;
class PriorityQueue{
  //Do not modify public part
  public:
    PriorityQueue();

    ~PriorityQueue();

    PriorityQueue operator*(const PriorityQueue& m) const;
    PriorityQueue operator+(const PriorityQueue& m) const;
    PriorityQueue& operator=(const PriorityQueue& m);

    void insert(int key, int val);
    int get_max() const;
    bool pop_max();

    void print_list() const;

  private:
    struct Cell{
      int key;
      int val;
      Cell* next;
    };
    Cell* head; //head of the list of elements
    
    //add all private methods you need here
    void add(int key, int val);
};

//Do not modify
PriorityQueue::PriorityQueue() {
  head = nullptr;
}

//Do not modify
PriorityQueue::~PriorityQueue() {
  Cell* pc = head;
  while (pc) {
    pc = pc->next;
    delete head;
    head = pc;
  }
}

//Do not modify
void PriorityQueue::print_list() const {
  Cell* pc = head;
  while (pc) {
    cout << "(" << pc->key << ", " << pc->val<<")"<<endl;
    pc = pc->next;
  }
}

//Do not modify
PriorityQueue::PriorityQueue(const PriorityQueue& m){
  head = nullptr;
  Cell*  pc = m.head;
  while (pc) {
    insert(pc->key, pc->val);
    pc = pc->next;
  }
}

//Exercise 1
void PriorityQueue::insert(int key, int val) {
    //if the list is empty and val is valid, add the first element
    if(!head){
        if(val > 0) add(key, val);
    }
    //if the first cell has the key
    else if(head->key == key){
        Cell* del = head;
        head = head->next;
        delete del;
        if(val > 0) add(key, val);
    }
    else{
        //parse the list till we get before the last pos or we find the key
        Cell* pt = head;
        while(pt->next and pt->next->key != key) pt = pt->next;
        //if key is present, delete it. if the value is valid, add it back.
        if(pt->next and pt->next->key == key){ 
            Cell* del = pt->next;
            pt->next = pt->next->next;
            delete del;
            if(val > 0) add(key, val);
        }
        //if the key is not found but value is valid, add it (no need to check anything)
        else if(val > 0) {
            add(key, val);
        }
    }
    return;
}

void PriorityQueue::add(int key, int val){
    if(!head)
        head = new Cell{key, val, nullptr};
    else if(head->val < val){
        Cell* add = new Cell{key, val, head};
        head = add;
    }
    else{
        Cell* pt = head;
        while (pt->next and pt->next->val >= val) pt = pt->next; 
        Cell* add = new Cell{key, val, pt->next};
        pt->next = add;
    }
}

int PriorityQueue::get_max() const {
    if(head) return head->key;
    else return 0;
}

//Exercise 2
bool PriorityQueue::pop_max() {
    if(head){
        Cell* del = head;
        head = head->next;
        delete del;
        return true;
    }
    else return false;
}

PriorityQueue PriorityQueue::operator+(const PriorityQueue& m) const {
    PriorityQueue res;
    res = m;
    Cell* pt = head;
    Cell* r_pt;
    while(pt){
        r_pt = res.head;
        while(r_pt and r_pt->key != pt->key) r_pt = r_pt->next;
        if(r_pt == nullptr) res.add(pt->key, pt->val);
        else if(pt->val < r_pt->val) res.insert(pt->key, pt->val);
        pt = pt->next;
    }
    return res;
}

PriorityQueue& PriorityQueue::operator=(const PriorityQueue& m) {
    if(head == m.head) return *this;
    Cell* pt;
    while(head){
        pt = head;
        head = head->next;
        delete pt;
    }
    Cell* m_pt = m.head;
    while(m_pt){
        add(m_pt->key, m_pt->val);
        m_pt = m_pt->next;
    }
  return *this;
}
