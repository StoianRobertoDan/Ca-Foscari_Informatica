#include<iostream>

class multiset{
public:
  multiset(); 
  multiset(const multiset& m);
  ~multiset();
  bool present(int e) const;
  void print() const;

  
  void add(int e);
  void remove(int e);

  bool operator<=(const multiset& m) const;
  bool operator>=(const multiset& m) const;
  multiset operator+(const multiset& m) const;
  
private:
  struct cell{
    int info;
    int quantity;
    cell* next;
  };
  cell* head;


};

//Do not modify
multiset::multiset() {
  head = nullptr;
}

//Do not modify
multiset::multiset(const multiset& m) {
  cell* pc = m.head;
  cell* pch = head = nullptr;

  while (pc) {
    cell* nc = new cell{pc->info, pc->quantity, nullptr};
    if (!pch) 
      head = nc;
    else 
      pch->next = nc;
    pch = nc;
    pc = pc->next;
  }
}

//Do not modify
multiset::~multiset() {
  cell* pc = head;
  while (pc) {
    head = pc->next;
    delete pc;
    pc = head;
  }
}

//Do not modify
void multiset::print() const {
  cell* pc = head;
  std::cout<<"{{";
  while(pc && pc->next) {
    std::cout<<"("<<pc->info<<", "<<pc->quantity<<"), ";
    pc = pc->next;
  }

  if (pc) 
    std::cout<<"("<<pc->info<<", "<<pc->quantity<<")}}"<<std::endl;
  else 
    std::cout<<"}}"<<std::endl;
}

//Decides if e is present in the multiset
bool multiset::present(int e) const {
    if(!head) return false;
    auto pt = head;
    while(pt){
        if(pt->info == e) return true;
        else pt = pt->next;
    }
    return false;
}

//Exercise 1
//Adds an element to the multi set. Elements must be add in the order
//of first insertion: if an element is present already, update its quantity
//otherwise add it to the tail of the list
void multiset::add(int e) {
    if(present(e)){
        auto pt = head;
        while(pt){
            if(pt->info == e) pt->quantity++;
            pt = pt->next;
        }
    }
    else{
        if(!head) head = new cell{e, 1, nullptr};
        else{
            auto pt = head;
            while(pt->next) pt = pt->next;
            auto c = new cell{e, 1, nullptr};
            pt->next = c;
        }
    }
} 


//Exercise 1
//Removes an element from the multiset. If the quantity is zero, the cell must be 
//completely deleted from the list.
//If the element is not present in the list, do nothing.
void multiset::remove(int e) {
    if(present(e)){
        if(head->info == e){
            if(head->quantity > 1) head->quantity--;
            else {
                auto tp = head;
                head = head->next;
                delete tp;
            }
        }
        else{
            auto pt = head;
            while(pt->next){
                if(pt->next->info == e){
                    if(pt->next->quantity > 1) pt->next->quantity--;
                    else{
                        auto tp = pt->next;
                        pt->next = pt->next->next;
                        delete tp;
                    }
                    return;
                }
                pt = pt->next;
            }
        }
    }
}

//Decides is a multiset is included in its parameter
//All the elements of the cuerrent multiset must be present in the
//parameter multiset with a quantity that is lower or equal than that of the parameter  
bool multiset::operator<=(const multiset& m) const {
    if(head and m.head){
        auto pt = head;
        bool minore = true;
        while(pt and minore){
            auto ptm = m.head;
            bool found = false;
            while(ptm and !found){
                if(pt->info == ptm->info){
                    found = true;
                    if(pt->quantity > ptm->quantity) minore = false;
                }
                ptm = ptm->next;
            }
            pt = pt->next;
        }
        return minore;
    }
    if(m.head and !head) return true;
    else return false;
}

bool multiset::operator>=(const multiset& m) const {
    if(head and m.head){
        auto pt = head;
        bool minore = true;
        while(pt and minore){
            auto ptm = m.head;
            bool found = false;
            while(ptm and !found){
                if(pt->info == ptm->info){
                    found = true;
                    if(pt->quantity < ptm->quantity) minore = false;
                }
                ptm = ptm->next;
            }
            pt = pt->next;
        }
        return minore;
    }
    if(head and !m.head) return true;
    else return false;
}

  }
